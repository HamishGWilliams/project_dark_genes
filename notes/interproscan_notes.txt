# Ran test script:
The test run failed because InterProScan rejected * characters in the protein FASTA. This is common in predicted protein files: * usually marks a stop codon from translation. InterProScan wants protein sequences without those stop characters.

First create a cleaned protein FASTA, rerun the 100-protein test, then run the full proteome using the cleaned FASTA.

The repository already treats 00_raw/equina_smart.rnam-trna.merged.ggf.curated.remredun.aa.fa as the predicted proteome and records that it contains 55,607 entries with matching CDS/transcript/GFF3 IDs. So we should keep the raw file unchanged and create a cleaned derivative for InterProScan.

mkdir -p 02_annotation/interproscan/input

python3 - <<'PY'
from pathlib import Path
from collections import Counter

inp = Path("00_raw/equina_smart.rnam-trna.merged.ggf.curated.remredun.aa.fa")
out = Path("02_annotation/interproscan/input/equina_smart.rnam-trna.merged.ggf.curated.remredun.aa.no_stop.fa")
report = Path("02_annotation/interproscan/input/equina_interproscan_fasta_cleaning_report.txt")

valid = set("ACDEFGHIKLMNPQRSTVWYBXZJUO")
seq_id = None
seq_chunks = []

n_seq = 0
n_with_star = 0
n_stars = 0
invalid_counts = Counter()
invalid_by_seq = {}

def flush():
    global n_seq, n_with_star, n_stars
    if seq_id is None:
        return

    seq = "".join(seq_chunks).replace(" ", "").replace("\t", "").upper()
    star_count = seq.count("*")
    cleaned = seq.replace("*", "")

    n_seq += 1
    if star_count:
        n_with_star += 1
        n_stars += star_count

    bad = sorted(set(cleaned) - valid)
    if bad:
        invalid_by_seq[seq_id] = "".join(bad)
        for char in bad:
            invalid_counts[char] += cleaned.count(char)

    fout.write(f">{seq_id}\n")
    for i in range(0, len(cleaned), 60):
        fout.write(cleaned[i:i+60] + "\n")

with inp.open() as fin, out.open("w") as fout:
    for line in fin:
        line = line.rstrip("\n")
        if line.startswith(">"):
            flush()
            seq_id = line[1:].strip()
            seq_chunks = []
        else:
            seq_chunks.append(line)
    flush()

with report.open("w") as r:
    r.write("InterProScan FASTA cleaning report\n")
    r.write("==================================\n\n")
    r.write(f"Input FASTA:  {inp}\n")
    r.write(f"Output FASTA: {out}\n\n")
    r.write(f"Sequences processed: {n_seq}\n")
    r.write(f"Sequences containing '*': {n_with_star}\n")
    r.write(f"Total '*' characters removed: {n_stars}\n\n")

    if invalid_counts:
        r.write("Non-standard characters remaining after '*' removal:\n")
        for char, count in sorted(invalid_counts.items()):
            r.write(f"  {char}: {count}\n")
        r.write("\nSequences with non-standard characters:\n")
        for sid, chars in list(invalid_by_seq.items())[:100]:
            r.write(f"  {sid}: {chars}\n")
    else:
        r.write("No non-standard amino-acid characters detected after '*' removal.\n")

print(report.read_text())
